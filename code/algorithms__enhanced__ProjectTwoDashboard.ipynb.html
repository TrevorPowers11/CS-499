# Set up the Jupyter version of Dash
from jupyter_dash import JupyterDash

# Import dashboard components
import dash_leaflet as dl
from dash import dcc, html, dash_table
from dash.dependencies import Input, Output
import plotly.express as px
import pandas as pd
import base64

# Import your existing CRUD module
from CRUD_Python_Module import AnimalShelter
from animal_index import AnimalIndex

JupyterDash.infer_jupyter_proxy_config()

# Connect to MongoDB through the CRUD module
username = 'REPLACE_WITH_LOCAL_CREDENTIAL'
password = 'REPLACE_WITH_LOCAL_CREDENTIAL'
db = AnimalShelter(username, password)

# Get the unfiltered animal records
index = AnimalIndex(db.read({}))
df = pd.DataFrame.from_records(index.records)

# MongoDB's _id field cannot be displayed in the Dash table
if "_id" in df.columns:
    df.drop(columns=["_id"], inplace=True)

# Load Grazioso Salvare logo
image_filename = "Grazioso Salvare Logo.png"
encoded_image = base64.b64encode(open(image_filename, "rb").read()).decode()

# Create dashboard
app = JupyterDash(__name__)

app.layout = html.Div([
    html.Center([
        html.Img(
            src="data:image/png;base64,{}".format(encoded_image),
            style={"width": "220px"}
        ),
        html.H1("Grazioso Salvare Animal Shelter Dashboard"),
        html.H3("Trevor Powers | CS 340")
    ]),

    html.Hr(),

    html.Div([
        html.H4("Select a Rescue Type:"),
        dcc.RadioItems(
            id="filter-type",
            options=[
                {"label": "Reset / All Animals", "value": "RESET"},
                {"label": "Water Rescue", "value": "WATER"},
                {"label": "Mountain or Wilderness Rescue", "value": "MOUNTAIN"},
                {"label": "Disaster or Individual Tracking", "value": "DISASTER"}
            ],
            value="RESET",
            labelStyle={"display": "block", "margin": "8px"}
        )
    ]),

    html.Div([
        dcc.Input(id="animal-search", type="text", placeholder="Exact animal ID"),
        dcc.Input(id="breed-search", type="text", placeholder="Exact breed"),
        dcc.Dropdown(id="sort-field", options=[
            {"label": label, "value": field} for label, field in [
                ("Animal ID", "animal_id"), ("Name", "name"),
                ("Breed", "breed"), ("Age in weeks", "age_upon_outcome_in_weeks")
            ]], value="animal_id", clearable=False),
        dcc.Checklist(id="sort-direction", options=[{"label": "Descending", "value": "desc"}], value=[])
    ]),

    html.Hr(),

    dash_table.DataTable(
        id="datatable-id",
        columns=[
            {"name": column, "id": column, "deletable": False, "selectable": True}
            for column in df.columns
        ],
        data=df.to_dict("records"),
        page_size=10,
        sort_action="none",
        filter_action="native",
        row_selectable="single",
        selected_rows=[0],
        style_table={"overflowX": "auto"},
        style_cell={
            "textAlign": "left",
            "minWidth": "110px",
            "width": "110px",
            "maxWidth": "180px",
            "whiteSpace": "normal"
        },
        style_header={
            "backgroundColor": "#1f5f3a",
            "color": "white",
            "fontWeight": "bold"
        }
    ),

    html.Br(),
    html.Hr(),

    html.Div(
        style={"display": "flex"},
        children=[
            html.Div(id="graph-id", style={"width": "50%"}),
            html.Div(id="map-id", style={"width": "50%"})
        ]
    )
])


# Index is built once. These controls reuse the same snapshot without a database round trip.
@app.callback(
    Output("datatable-id", "data"),
    Input("filter-type", "value"), Input("animal-search", "value"),
    Input("breed-search", "value"), Input("sort-field", "value"),
    Input("sort-direction", "value")
)
def update_dashboard(filter_type, animal_id, breed, sort_by, direction):
    return index.search(filter_type or "RESET", animal_id or "", breed or "",
                        sort_by or "animal_id", "desc" in (direction or []))


# Create a dynamic breed chart from the table data
@app.callback(
    Output("graph-id", "children"),
    Input("datatable-id", "derived_virtual_data")
)
def update_graphs(view_data):

    if not view_data:
        return html.P("No animal records match this filter.")

    chart_df = pd.DataFrame(view_data)
    breed_counts = chart_df["breed"].value_counts().head(10).reset_index()
    breed_counts.columns = ["Breed", "Count"]

    figure = px.bar(
        breed_counts,
        x="Breed",
        y="Count",
        title="Top Breeds in Current Results",
        color="Count",
        color_continuous_scale="Greens"
    )

    figure.update_layout(xaxis_tickangle=-45)

    return dcc.Graph(figure=figure)


# Highlight selected table column
@app.callback(
    Output("datatable-id", "style_data_conditional"),
    Input("datatable-id", "selected_columns")
)
def update_styles(selected_columns):
    return [
        {
            "if": {"column_id": column},
            "background_color": "#D2F3FF"
        }
        for column in selected_columns
    ]


# Update the map based on the selected table row
@app.callback(
    Output("map-id", "children"),
    [
        Input("datatable-id", "derived_virtual_data"),
        Input("datatable-id", "derived_virtual_selected_rows")
    ]
)
def update_map(view_data, selected_rows):

    if not view_data:
        return html.P("No map data is available for this filter.")

    map_df = pd.DataFrame(view_data)

    row = 0
    if selected_rows and len(selected_rows) > 0:
        row = selected_rows[0]

    if row >= len(map_df):
        row = 0

    latitude = map_df.iloc[row]["location_lat"]
    longitude = map_df.iloc[row]["location_long"]
    breed = map_df.iloc[row]["breed"]
    animal_name = map_df.iloc[row]["name"]

    return [
        dl.Map(
            style={"width": "100%", "height": "500px"},
            center=[latitude, longitude],
            zoom=10,
            children=[
                dl.TileLayer(),
                dl.Marker(
                    position=[latitude, longitude],
                    children=[
                        dl.Tooltip(breed),
                        dl.Popup([
                            html.H4("Animal Information"),
                            html.P("Name: " + str(animal_name)),
                            html.P("Breed: " + str(breed))
                        ])
                    ]
                )
            ]
        )
    ]

app.run_server(mode="external")
