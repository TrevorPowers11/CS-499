import importlib
import CRUD_Python_Module

# Reload the module to use the latest saved version
importlib.reload(CRUD_Python_Module)

from CRUD_Python_Module import AnimalShelter

# Connect using the MongoDB user created for the AAC database
shelter = AnimalShelter('REPLACE_WITH_LOCAL_CREDENTIAL', 'REPLACE_WITH_LOCAL_CREDENTIAL')

print("CRUD module imported successfully.")
print("AnimalShelter object created successfully.")

# Remove any leftover test records from previous runs
shelter.delete({"animal_id": "PROJECT_ONE_TREVOR_TEST"})

# Test Create
test_document = {
    "animal_id": "PROJECT_ONE_TREVOR_TEST",
    "name": "Trevor Test Animal",
    "animal_type": "Dog",
    "breed": "Labrador Retriever Mix",
    "color": "Black",
    "age_upon_outcome": "2 years",
    "outcome_type": "Adoption"
}

create_result = shelter.create(test_document)
print("\nCREATE result:", create_result)

# Test Read
query = {"animal_id": "PROJECT_ONE_TREVOR_TEST"}
read_result = shelter.read(query)

print("\nREAD returned a:", type(read_result))
print("Records found:", len(read_result))
print("Record before update:", read_result)

# Test Update
update_result = shelter.update(
    query,
    {
        "name": "Trevor Updated Animal",
        "outcome_type": "Transfer"
    }
)

print("\nUPDATE records modified:", update_result)

updated_records = shelter.read(query)
print("Record after update:", updated_records)

# Test Delete
delete_result = shelter.delete(query)
print("\nDELETE records removed:", delete_result)

remaining_records = shelter.read(query)
print("Records remaining after delete:", len(remaining_records))
