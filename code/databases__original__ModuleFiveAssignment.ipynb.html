# DASH Framework for Jupyter
from jupyter_dash import JupyterDash
from dash import dcc, html
from dash.dependencies import Input, Output
JupyterDash.infer_jupyter_proxy_config()

import urllib.parse

# Import existing Project One CRUD module
from CRUD_Python_Module import AnimalShelter

# Build App
app = JupyterDash(__name__)

app.layout = html.Div([
    html.H1("Module 5 Assignment - Trevor Powers"),

    dcc.Input(
        id="input_user",
        type="text",
        placeholder="Username"
    ),

    dcc.Input(
        id="input_passwd",
        type="password",
        placeholder="Password"
    ),

    html.Button("Submit", id="submit-val", n_clicks=0),

    html.Hr(),

    html.Div(id="query-out", style={"whiteSpace": "pre-line"})
])

# Dashboard callback
@app.callback(
    Output("query-out", "children"),
    [
        Input("input_user", "value"),
        Input("input_passwd", "value"),
        Input("submit-val", "n_clicks")
    ]
)
def update_figure(inputUser, inputPass, n_clicks):

    if n_clicks > 0:
        try:
            username = urllib.parse.quote_plus(inputUser)
            password = urllib.parse.quote_plus(inputPass)

            shelter = AnimalShelter(username, password)

            results = shelter.read({
                "animal_type": "Dog",
                "name": "Lucy"
            })

            if results:
                return "\n\n".join(str(result) for result in results)

            return "No matching records found."

        except Exception as error:
            return "Error: " + str(error)

    return "Enter your username and password, then click Submit."


app.run_server(mode="inline")
